cmake_minimum_required(VERSION 3.8)
# Accept the legacy Python option used by existing ROS build instructions.
if(DEFINED PYTHON_EXECUTABLE)
  if(DEFINED Python3_EXECUTABLE AND NOT "${Python3_EXECUTABLE}" STREQUAL "${PYTHON_EXECUTABLE}")
    message(FATAL_ERROR "PYTHON_EXECUTABLE and Python3_EXECUTABLE must select the same interpreter")
  endif()
  set(Python3_EXECUTABLE "${PYTHON_EXECUTABLE}")
endif()
project(bxi_motor_control LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(bxi_motor_msgs REQUIRED)
get_filename_component(DRIVER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(ETHERLAB_ROOT "" CACHE PATH "EtherLab prefix")
find_path(ETHERLAB_INCLUDE_DIR ecrt.h HINTS ${ETHERLAB_ROOT} /usr/local/etherlab /opt/etherlab PATH_SUFFIXES include)
find_library(ETHERLAB_LIBRARY ethercat HINTS ${ETHERLAB_ROOT} /usr/local/etherlab /opt/etherlab PATH_SUFFIXES lib lib64)
if(NOT ETHERLAB_INCLUDE_DIR OR NOT ETHERLAB_LIBRARY)
  message(FATAL_ERROR "ROS motor backend requires EtherLab headers and library")
endif()
add_library(motor_backend STATIC src/backend.c "${DRIVER_ROOT}/src/lib/bxi_motor_comm.c")
target_compile_definitions(motor_backend PRIVATE HAVE_ETHERLAB=1)
target_include_directories(motor_backend PRIVATE "${DRIVER_ROOT}/src" "${DRIVER_ROOT}/src/lib" "${DRIVER_ROOT}/lib" "${ETHERLAB_INCLUDE_DIR}")
target_compile_options(motor_backend PRIVATE -Wall -Wextra -Wno-unused-function)
target_link_libraries(motor_backend PUBLIC "${DRIVER_ROOT}/lib/libbxi_pci_drv.a" "${ETHERLAB_LIBRARY}" m pthread)
add_executable(motor_node src/node.cpp)
target_compile_features(motor_node PRIVATE cxx_std_17)
ament_target_dependencies(motor_node rclcpp std_msgs bxi_motor_msgs)
target_link_libraries(motor_node motor_backend)
install(TARGETS motor_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.py")
install(FILES "${DRIVER_ROOT}/config/motor_console.yaml" "${DRIVER_ROOT}/config/flash_plan_default.yaml" "${DRIVER_ROOT}/config/kaixuan_modules.json" DESTINATION share/${PROJECT_NAME}/config)
ament_package()
